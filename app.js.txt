const PRODUCTS = [
  {id:'banshee', title:'Banshee VIP', price:12.99},
  {id:'gtx', title:'GTX VIP', price:24.99}
];

function formatPrice(p){ return '$' + p.toFixed(2); }

function render(){
  const el = document.getElementById('products');
  PRODUCTS.forEach(p=>{
    const card = document.createElement('div');
    card.className='card';
    card.innerHTML = `<h3>${p.title}</h3><div>${formatPrice(p.price)}</div><button class="btn">Buy</button>`;
    el.appendChild(card);
  });
}
document.addEventListener('DOMContentLoaded', render);
